"""16 kHz mono RAM-only capture. No stream exists until intentional opening."""
from threading import RLock
import time
from safety import Refusal


class Audio:
    def __init__(self, post, state, *, max_seconds=15, silence_seconds=0.7, clock=time.monotonic):
        self.post, self.state, self.clock = post, state, clock
        self.max_seconds, self.silence_seconds = max_seconds, silence_seconds
        self.stream = None
        self.lock = RLock()
        self.frames, self.speech, self.silence, self.stopping = [], 0, 0, False
        self.live, self.speaking, self.conversation = False, False, False

    def open(self, conversation=False):
        with self.lock:
            if self.speaking:
                raise Refusal('Microphone cannot open during speech.')
            if self.live:
                return
            import sounddevice as sd
            import webrtcvad
            self.vad = webrtcvad.Vad(2)
            self.frames, self.speech, self.silence, self.stopping = [], 0, 0, False
            self.conversation = conversation
            self.chime(True)
            self.state('listening')
            self.live = True
            try:
                self.stream = sd.RawInputStream(samplerate=16000, channels=1, dtype='int16',
                                                blocksize=480, callback=self._capture)
                self.stream.start()
            except Exception:
                self.live = False
                if self.stream:
                    self.stream.close()
                self.stream = None
                self.state('idle')
                raise

    def _capture(self, data, frames, timing, status):
        with self.lock:
            if not self.live or self.speaking or self.stopping:
                return
            if status or frames != 480:
                self.stopping = True
                self.post('audio_error')
                return
            chunk = bytes(data)
            self.frames.append(chunk)
            voiced = self.vad.is_speech(chunk, 16000)
            if voiced:
                self.speech += 1
                self.silence = 0
            else:
                self.silence += 1
            if not self.speech and len(self.frames) > 10:
                self.frames = self.frames[-10:]
            # Active audio frame callbacks are not idle polling.
            if len(self.frames) * .03 >= self.max_seconds or (
                self.conversation and self.speech >= 5 and self.silence * .03 >= self.silence_seconds):
                self.stopping = True
                self.post('utterance')

    def close(self, *, discard=False, notice=True):
        with self.lock:
            was_live, self.live = self.live, False
            stream, self.stream = self.stream, None
        if stream:
            stream.abort()
            stream.close()
        with self.lock:
            data = b''.join(self.frames) if not discard and self.speech >= 5 else b''
            self.frames.clear()
            self.speech = self.silence = 0
        if was_live:
            self.state('idle')
            if notice:
                self.chime(False)
        return data

    def begin_speech(self):
        self.close(discard=True)
        with self.lock:
            self.speaking = True
        self.state('speaking')

    def end_speech(self):
        with self.lock:
            self.speaking = False
        self.state('idle')

    @staticmethod
    def chime(opening):
        import numpy as np
        import sounddevice as sd
        samples = np.arange(1600, dtype=np.float32) / 16000
        wave = .08 * np.sin(2 * np.pi * (880 if opening else 440) * samples)
        sd.play(wave, 16000, blocking=True)

"""Append-only, redacted hash chain. This is not an authenticated log."""
import hashlib
import json
import os
from threading import RLock
from datetime import datetime, timezone
from safety import Refusal, no_links


def digest(value):
    return hashlib.sha256(json.dumps(value, sort_keys=True, separators=(',', ':')).encode()).hexdigest()


class Audit:
    def __init__(self, path, max_bytes=4_000_000, mode=lambda: 'OFFLINE'):
        self.path = no_links(path)
        self.max_bytes = max_bytes
        self.head = self.verify()
        self.lock, self.mode = RLock(), mode

    def verify(self):
        head = '0' * 64
        if not self.path.exists():
            return head
        no_links(self.path)
        if self.path.stat().st_nlink != 1 or self.path.stat().st_size > self.max_bytes:
            raise Refusal('Audit file exceeds limits or is a hard link.')
        try:
            with self.path.open(encoding='utf-8') as stream:
                for line in stream:
                    row = json.loads(line)
                    recorded = row.pop('hash')
                    if row['previous'] != head or digest(row) != recorded:
                        raise ValueError('chain')
                    head = recorded
        except (ValueError, KeyError, TypeError) as exc:
            raise Refusal('Audit integrity check failed.') from exc
        return head

    def record(self, tool, args, result, tier=0):
        with self.lock:
            return self._record(tool, args, result, tier)

    def _record(self, tool, args, result, tier):
        if self.verify() != self.head:
            raise Refusal('Audit changed during this session.')
        row = {'timestamp': datetime.now(timezone.utc).isoformat(),
               'tool': tool, 'args': {k: '[redacted]' for k in args},
               'result': result, 'mode': self.mode(), 'tier': tier, 'previous': self.head}
        checksum = digest(row)
        line = json.dumps({**row, 'hash': checksum}, sort_keys=True) + '\n'
        size = self.path.stat().st_size if self.path.exists() else 0
        if size + len(line.encode()) > self.max_bytes:
            raise Refusal('Audit capacity reached; refusing further actions.')
        with self.path.open('a', encoding='utf-8') as stream:
            stream.write(line)
            stream.flush()
            os.fsync(stream.fileno())
        self.head = checksum

from pathlib import Path
from urllib.parse import quote

from schemas import checked
from safety import Refusal, verify_model, no_links
from lifecycle import OnlineMode

class Chrome:
    def __init__(self, config, profile_root, notice, audit):
        self.config, self.profile_root, self.audit = config, Path(profile_root), audit
        self.process, self.pid, self.job = None, None, None
        self.mode = OnlineMode(self._close, notice, seconds=config.get('seconds', 900))

    @checked('enable_online')
    def enable_online(self, args):
        if not self.config.get('enabled'):
            raise Refusal('Online mode is disabled in configuration, sir.')
        check_chrome_policies(self.config['hosts'])
        verify_model(self.config['path'], self.config['sha256'], 500_000_000)
        self.mode.enable()
        return 'Restricted browsing is ready, sir.'

    def _require_online(self):
        self.mode.tick()
        if not self.mode.active:
            raise Refusal('Online mode has expired or is disabled, sir.')
        check_chrome_policies(self.config['hosts'])
        if self.profile_root.exists():
            used = sum(path.stat().st_size for path in self.profile_root.rglob('*') if path.is_file())
            if used >= 64_000_000:
                self.mode.close()
                raise Refusal('Retained browser profiles reached their disk allowance; Online mode closed.')

    @checked('open_chrome')
    def open_chrome(self, args):
        self._require_online()
        url = safe_url(args['url'], self.config['hosts'])
        exe = verify_model(self.config['path'], self.config['sha256'], 500_000_000)
        if self.job:
            self._close()
        import subprocess
        import win32api
        import win32con
        import win32job
        import win32process
        profile = no_links(self.profile_root / uuid4().hex)
        profile.mkdir(parents=True, exist_ok=False)
        argv = [str(exe), '--user-data-dir=' + str(profile), '--no-first-run', '--no-default-browser-check',
                '--disable-extensions', '--disable-background-mode', '--disable-sync', '--incognito',
                '--disk-cache-size=16777216', '--media-cache-size=16777216', url]
        job = win32job.CreateJobObject(None, None)
        limits = win32job.QueryInformationJobObject(job, win32job.JobObjectExtendedLimitInformation)
        limits['BasicLimitInformation']['LimitFlags'] = win32job.JOB_OBJECT_LIMIT_KILL_ON_JOB_CLOSE
        win32job.SetInformationJobObject(job, win32job.JobObjectExtendedLimitInformation, limits)
        process = thread = None
        try:
            process, thread, pid, _ = win32process.CreateProcess(str(exe), subprocess.list2cmdline(argv), None, None,
                False, win32con.CREATE_SUSPENDED | win32con.CREATE_NO_WINDOW, None, str(exe.parent), win32process.STARTUPINFO())
            win32job.AssignProcessToJobObject(job, process)
            self.job, self.process, self.pid = job, process, pid
            win32process.ResumeThread(thread)
        except Exception:
            # Closing the owned job stops only its processes; no generic process killer exists.
            job.Close()
            if process:
                process.Close()
            raise
        finally:
            if thread:
                thread.Close()
        return 'Opening the approved page in restricted Chrome, sir.'

    @checked('web_search')
    def web_search(self, args):
        self._require_online()
        url = self.config['search_url'] + quote(args['query'], safe='')
        return self.open_chrome({'url': url})

    def _close(self):
        if self.job:
            self.job.Close()
            self.job = None
        if self.process:
            self.process.Close()
            self.process = None
        self.pid = None
        self.audit.record('close_chrome', {}, 'closed')

    @checked('close_chrome')
    def close_chrome(self, args):
        self.mode.close()
        return 'Offline mode restored, sir.'

    @checked('type_text')
    def type_text(self, args):
        self._require_online()
        if not self.config.get('input_control') or not self.pid or not self.job:
            raise Refusal('Browser input control is disabled, sir.')
        import win32gui
        import win32process
        from pynput.keyboard import Controller
        text = safe_typing(args['text'])
        keyboard = Controller()
        # No Enter/Tab is sent: form submission is deliberately not a tool.
        for char in text:
            self._require_online()
            if win32process.GetWindowThreadProcessId(win32gui.GetForegroundWindow())[1] != self.pid:
                raise Refusal('Focus left my Chrome window; typing stopped, sir.')
            keyboard.type(char)
        return 'Text entered without submitting the form, sir.'

import re
from urllib.parse import urlsplit
from safety import Refusal


def safe_url(url, hosts):
    if not isinstance(url, str) or len(url) > 2048 or any(ord(c) < 33 for c in url):
        raise Refusal('Invalid browser URL.')
    parsed = urlsplit(url)
    if (parsed.scheme != 'https' or parsed.hostname not in hosts or parsed.username or parsed.password
            or parsed.port not in (None, 443) or '\\' in url):
        raise Refusal('Only approved HTTPS domains are permitted, sir.')
    if re.search(r'(login|signin|checkout|payment|download|webstore|oauth)', url, re.I):
        raise Refusal('That browser destination is outside my permissions.')
    return url


def safe_typing(text):
    if not isinstance(text, str) or not 0 < len(text) <= 500:
        raise Refusal('Text is too long.')
    if re.search(r'(password|passcode|card|cvv|otp|one.?time|payment|secret|token|pin\b|\d{4,})', text, re.I):
        raise Refusal('Sensitive text cannot be entered, sir.')
    if any(mark in text for mark in (';', '|', '`', '$', '<', '>', '{', '}', '\\', '\n', '\r')):
        raise Refusal('Shell-like or control text is not permitted.')
    return ''.join(c for c in text if ord(c) >= 32 and ord(c) != 127)


def check_chrome_policies(hosts):
    import winreg
    base = r'SOFTWARE\Policies\Google\Chrome'
    scalar = {'DownloadRestrictions': 3, 'PasswordManagerEnabled': 0,
              'BrowserSignin': 0, 'SyncDisabled': 1, 'IncognitoModeAvailability': 1,
              'DeveloperToolsAvailability': 2}
    flags = winreg.KEY_READ | winreg.KEY_WOW64_64KEY
    try:
        with winreg.OpenKey(winreg.HKEY_LOCAL_MACHINE, base, 0, flags) as key:
            for name, expected in scalar.items():
                if winreg.QueryValueEx(key, name)[0] != expected:
                    raise Refusal('Required Chrome policy is missing.')
        required = {'ExtensionInstallBlocklist': {'*'}, 'URLBlocklist': {'*'},
                    'URLAllowlist': {'https://' + host for host in hosts}}
        for name, expected in required.items():
            with winreg.OpenKey(winreg.HKEY_LOCAL_MACHINE, base + '\\' + name, 0, flags) as key:
                count = winreg.QueryInfoKey(key)[1]
                actual = {winreg.EnumValue(key, i)[1] for i in range(count)}
                if actual != expected:
                    raise Refusal('Chrome URL or extension policies do not match the approved list.')
    except OSError as exc:
        raise Refusal('Machine Chrome policies must be provisioned and verified first.') from exc

from pathlib import Path
import yaml
from jsonschema import Draft202012Validator
from safety import Refusal
from health import PROFILES


def load_config(root):
    with (root / 'config.yaml').open(encoding='utf-8') as stream:
        config = yaml.safe_load(stream)
    schema = {'type': 'object', 'required': ['profile', 'dry_run', 'model', 'stt', 'tts', 'security', 'apps', 'browser'],
              'properties': {
                  'profile': {'enum': list(PROFILES)}, 'dry_run': {'type': 'boolean'},
                  'max_calls_per_minute': {'type': 'integer', 'minimum': 1, 'maximum': 60},
                  'failure_limit': {'type': 'integer', 'minimum': 1, 'maximum': 10},
                  'stt_confidence': {'type': 'number', 'minimum': .75, 'maximum': 1},
                  'confirmation_seconds': {'type': 'integer', 'minimum': 1, 'maximum': 8},
                  'conversation_seconds': {'type': 'integer', 'minimum': 5, 'maximum': 60},
                  'idle_exit_seconds': {'type': 'integer', 'minimum': 30, 'maximum': 600},
                  'model_idle_seconds': {'type': 'integer', 'minimum': 5, 'maximum': 60}}}
    Draft202012Validator(schema).validate(config)
    if any(config.get('optional', {}).values()):
        raise Refusal('Optional wake word, identity verification and persistent memory are not enabled in this build.')
    for key in ('model', 'stt'):
        item = config[key]
        if not isinstance(item.get('threads'), int) or not 1 <= item['threads'] <= 8:
            raise Refusal('Inference thread count is outside the safe range.')
    if not 1024 <= config['model']['context'] <= 2048 or not 1 <= config['model']['max_tokens'] <= 256:
        raise Refusal('Model context or reply token cap is outside the safe range.')
    for item, field in [(config['model'], 'path'), (config['stt'], 'path'),
                         (config['tts'], 'path'), (config['tts'], 'config_path')]:
        path = Path(item[field]).expanduser()
        item[field] = str(path if path.is_absolute() else root / path)
    for key in ('approved_folder', 'log_dir'):
        config[key] = str(Path(config[key]).expanduser())
    return config

"""Read-only readiness report. Does not open audio streams, launch apps, or download."""
from importlib import metadata
import json
from pathlib import Path
import platform
import sys
from safety import verify_model


def report(root, config):
    packages = {}
    for line in (root / 'requirements.in').read_text().splitlines():
        if '==' not in line or line.startswith('#'):
            continue
        package, rest = line.split('==', 1)
        expected = rest.split(';')[0].strip()
        try:
            actual = metadata.version(package)
        except metadata.PackageNotFoundError:
            actual = None
        packages[package] = {'expected': expected, 'installed': actual, 'matches': actual == expected}
    models = {}
    for key in ('model', 'stt', 'tts'):
        item = config[key]
        try:
            verify_model(item['path'], item['sha256'], item['max_bytes'])
            models[key] = 'verified'
        except Exception as exc:
            models[key] = str(exc)
    result = {'python': platform.python_version(), 'executable': sys.executable,
              'dedicated_venv': sys.prefix != sys.base_prefix,
              'hashed_lock_present': (root / 'requirements.txt').is_file(),
              'profile': config['profile'], 'packages': packages, 'models': models,
              'online_enabled': config['browser']['enabled'],
              'production_ready': False,
              'blockers': ['OS setup and real voice/model validation pending',
                           'Outbound self-test exception and implementation unresolved',
                           'Native stack and model footprint must fit the hard profile budget']}
    print(json.dumps(result, indent=2))
    return result

from jsonschema import Draft202012Validator
from safety import Refusal, user_text
from tools.registry import call_schema


class Assistant:
    def __init__(self, intent, policy, llm=None, memory=None, chat_sink=None):
        self.intent, self.policy, self.llm = intent, policy, llm
        self.memory, self.chat_sink = memory, chat_sink
        self.streamed = False

    def handle(self, text, *, source='user'):
        try:
            self.streamed = False
            if self.policy.failures >= self.policy.failure_limit:
                raise Refusal('Safety lock engaged. Restart required, sir.')
            if source != 'user':
                return self.policy.refuse('External content cannot originate actions, sir.')
            try:
                text = user_text(text)
            except Refusal as exc:
                return self.policy.refuse(str(exc))
            if self.policy.pending:
                return self.policy.confirm(text)
            call = self.intent.match(text)
            if not call and text.strip().lower() in ('turn it down a bit', 'now turn it down a bit', 'turn it up a bit'):
                if self.memory is None or self.memory.last_volume is None:
                    return 'Please specify a volume percentage, sir.'
                delta = 5 if 'up' in text else -5
                call = {'tool': 'set_volume', 'args': {'level': max(0, min(100, self.memory.last_volume + delta))}}
            if call:
                return self._execute(call)
            if self.llm is None:
                return 'Local conversation is unavailable until a verified model is configured, sir.'
            call = self.llm.route(text)
            Draft202012Validator(call_schema(self.policy.registry)).validate(call)
            if call.get('route') == 'chat':
                # The return value is display-only, even if it resembles a tool call.
                if self.chat_sink and hasattr(self.llm, 'stream_chat'):
                    self.streamed = True
                    parts = []
                    for sentence in self.llm.stream_chat(text, self.memory.messages() if self.memory else []):
                        self.chat_sink(sentence)
                        parts.append(sentence)
                    answer = ' '.join(parts)
                else:
                    answer = self.llm.chat(text)
                if self.memory:
                    self.memory.remember(text, answer)
                return answer
            return self._execute(call)
        except Refusal as exc:
            self.streamed = False
            return str(exc)
        except Exception:
            self.streamed = False
            self.policy.failures += 1
            self.policy.audit.record('request', {}, 'error')
            return 'The request failed safely, sir.'

    def _execute(self, call):
        result = self.policy.dispatch(call)
        if self.memory and call['tool'] == 'set_volume' and not self.policy.dry_run:
            self.memory.last_volume = call['args']['level']
        return result

"""Manual-only, pinned HTTPS provisioning. Never imported by the assistant."""
import argparse
import hashlib
import json
from pathlib import Path
import re
import shutil
import time
from urllib.parse import urlsplit
from urllib.request import HTTPRedirectHandler, HTTPSHandler, ProxyHandler, Request, build_opener
from safety import Refusal, no_links


class PinnedRedirects(HTTPRedirectHandler):
    def __init__(self, hosts):
        self.hosts, self.count = set(hosts), 0

    def redirect_request(self, req, fp, code, msg, headers, newurl):
        self.count += 1
        validate_url(newurl, self.hosts)
        if self.count > 5:
            raise Refusal('Too many model redirects.')
        return super().redirect_request(req, fp, code, msg, headers, newurl)


def validate_url(url, hosts):
    parsed = urlsplit(url)
    if parsed.scheme != 'https' or parsed.hostname not in hosts or parsed.port not in (None, 443) or parsed.username or parsed.password:
        raise Refusal('Model URL or redirect is not approved HTTPS.')


def provision(entry, root, *, opener=None):
    validate_entry(entry)
    return _provision(entry, root, opener=opener)


def validate_entry(entry):
    required = {'url', 'sha256', 'filename', 'max_bytes', 'hosts'}
    if set(entry) != required or not re.fullmatch('[0-9a-f]{64}', entry['sha256'] or ''):
        raise Refusal('A complete trusted model manifest and SHA256 are required.')
    if not re.fullmatch(r'[A-Za-z0-9][A-Za-z0-9_.-]{0,100}\.(gguf|bin|onnx|json)', entry['filename']):
        raise Refusal('Invalid model filename.')
    if not isinstance(entry['max_bytes'], int) or not 0 < entry['max_bytes'] <= 500_000_000:
        raise Refusal('Provisioning is capped to the ULTRA-LITE per-file allowance.')
    validate_url(entry['url'], entry['hosts'])


def _provision(entry, root, *, opener=None):
    root = no_links(root)
    root.mkdir(parents=True, exist_ok=True)
    path = no_links(root / entry['filename'])
    if path.exists():
        raise Refusal('The destination exists; it will not be overwritten.')
    if shutil.disk_usage(root).free < entry['max_bytes'] + 100_000_000:
        raise Refusal('Insufficient free disk space.')
    opener = opener or build_opener(ProxyHandler({}), PinnedRedirects(entry['hosts']), HTTPSHandler())
    digest, size, started = hashlib.sha256(), 0, time.monotonic()
    # CREATE_NEW semantics; interrupted or mismatched files are retained, never loaded without hash validation.
    with opener.open(Request(entry['url'], headers={'User-Agent': 'JARVIS-manual-provisioner'}), timeout=15) as response:
        validate_url(response.geturl(), entry['hosts'])
        with path.open('xb') as target:
            while True:
                chunk = response.read(65536)
                if not chunk:
                    break
                size += len(chunk)
                if size > entry['max_bytes'] or time.monotonic() - started > 900:
                    raise Refusal('Model download exceeded its size or time cap; incomplete file retained.')
                target.write(chunk)
                digest.update(chunk)
    if digest.hexdigest() != entry['sha256']:
        raise Refusal('Checksum mismatch; untrusted file retained and runtime will refuse it.')
    return path


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('manifest', type=Path)
    parser.add_argument('--output', type=Path, default=Path(__file__).resolve().parent / 'models')
    args = parser.parse_args()
    manifest = json.loads(args.manifest.read_text(encoding='utf-8'))

def normalize(text):
    text = text.lower().strip().rstrip('?.!')
    text = re.sub(r'^please\s+', '', text)
    return re.sub(r'[, ]+(please|sir)$', '', text).strip()


class Intent:
    def __init__(self, commands):
        self.phrases = {normalize(phrase): tool for tool, phrases in commands.items() for phrase in phrases}

    def match(self, text):
        original = user_text(text).strip()
        text = normalize(original)
        if text in self.phrases:
            return {'tool': self.phrases[text], 'args': {}}
        # Recognize later-phase syntax but let the policy refuse unavailable tools.
        for pattern, tool, key in [(r'open (.+)', 'open_app', 'app'),
                                   (r'volume (\d{1,3})', 'set_volume', 'level')]:
            matched = re.fullmatch(pattern, text)
            if matched:
                value = matched[1]
                return {'tool': tool, 'args': {key: int(value) if key == 'level' else value}}
        if text == 'mute':
            return {'tool': 'mute', 'args': {}}
        from schemas import SCHEMAS
        aliases = {'forget that': 'forget_conversation', 'stop speaking': 'stop_speaking',
                   'what did you just do': 'what_did_you_do', 'list notes': 'list_folder',
                   'take a screenshot': 'take_screenshot', 'lock screen': 'lock_screen',
                   'pause music': 'media_play_pause', 'play music': 'media_play_pause',
                   'next track': 'media_next', 'previous track': 'media_prev',
                   'go online': 'enable_online', 'go offline': 'close_chrome',
                   'undo last note action': 'undo_last_note_action'}
        if text in aliases:
            return {'tool': aliases[text], 'args': {}}
        for pattern, tool, key in [(r'close ([a-z][a-z0-9_-]*)', 'close_app', 'app'),
                                   (r'brightness (\d{1,3})', 'set_brightness', 'level'),
                                   (r'read (.+\.txt)', 'read_file', 'name')]:
            matched = re.fullmatch(pattern, text)
            if matched:
                return {'tool': tool, 'args': {key: int(matched[1]) if key == 'level' else matched[1]}}
        matched = re.fullmatch(r'(create|append) note (.+\.txt): (.+)', original, re.I)
        if matched:
            return {'tool': 'create_note' if matched[1].lower() == 'create' else 'append_note',
                    'args': {'name': matched[2], 'text': matched[3]}}
        matched = re.fullmatch(r'(?:set )?timer (\d+) (seconds?|minutes?)(?: for (.+))?', text)
        if matched:
            return {'tool': 'set_timer', 'args': {'seconds': int(matched[1]) * (60 if matched[2].startswith('minute') else 1),
                                                'label': matched[3] or 'Your timer has finished.'}}
        return None

    def suggest(self, text):
        # Fuzzy matches are suggestions only. Never execute a near match.
        from rapidfuzz import process, fuzz
        match = process.extractOne(normalize(user_text(text)), self.phrases, scorer=fuzz.ratio, score_cutoff=85)
        return match[0] if match else None

class Conversation:
    def __init__(self, open_mic, close_mic, *, timeout=30, clock=time.monotonic):
        self.open_mic, self.close_mic, self.timeout, self.clock = open_mic, close_mic, timeout, clock
        self.enabled, self.deadline = False, 0

    def toggle(self, *, on_battery=False, allow_battery=False):
        if self.enabled:
            self.close()
        else:
            if on_battery and not allow_battery:
                raise Refusal('Conversation mode is disabled on battery, sir.')
            self.enabled = True
            self.touch()
            try:
                self.open_mic()
            except Exception:
                self.enabled = False
                raise

    def touch(self):
        self.deadline = self.clock() + self.timeout

    def tick(self):
        if self.enabled and self.clock() >= self.deadline:
            self.close()
            return True
        return False

    def close(self):
        self.enabled = False
        self.close_mic()

import json
from jsonschema import Draft202012Validator
from safety import Refusal, verify_model
from tools.registry import call_schema


class LocalLLM:
    def __init__(self, config, persona, registry=None, cancelled=lambda: False):
        self.config, self.persona, self.model = config, persona, None
        self.schema = call_schema(registry) if registry is not None else call_schema()
        self.cancelled = cancelled

    def _load(self):
        if self.model is None:
            path = verify_model(self.config['path'], self.config['sha256'], self.config['max_bytes'])
            from llama_cpp import Llama
            self.model = Llama(model_path=str(path), n_ctx=self.config['context'],
                               n_threads=self.config['threads'], n_gpu_layers=0,
                               use_mmap=True, verbose=False)
        return self.model

    def route(self, text):
        from llama_cpp import LlamaGrammar
        schema = self.schema
        grammar = LlamaGrammar.from_json_schema(json.dumps(schema), verbose=False)
        choices = []
        for branch in schema['oneOf']:
            properties = branch['properties']
            name = properties['tool']['const']
            if name != 'none':
                choices.append(name + '(' + ', '.join(properties['args'].get('properties', {})) + ')')
        messages = [{'role': 'system', 'content': 'Route only the current user request to one tool and args. '
                    'Otherwise emit {"tool":"none","route":"chat"}. Tools: ' + '; '.join(choices)},
                    {'role': 'user', 'content': text}]
        self._check_context(messages, 128)
        reply = self._load().create_chat_completion(
            messages=messages, grammar=grammar, temperature=0, max_tokens=128)
        try:
            call = json.loads(reply['choices'][0]['message']['content'])
            Draft202012Validator(schema).validate(call)
        except Exception as exc:
            raise Refusal('The model produced an invalid route, sir.') from exc
        return call

    def chat(self, text):
        # No tool schema, dispatcher, callbacks, or policy object is exposed here.
        messages = [{'role': 'system', 'content': self.persona}, {'role': 'user', 'content': text}]
        self._check_context(messages, self.config['max_tokens'])
        response = self._load().create_chat_completion(
            messages=messages,
            temperature=0.4, max_tokens=self.config['max_tokens'])
        return str(response['choices'][0]['message']['content'])[:2000]

    def stream_chat(self, text, history=()):
        import re
        messages = [{'role': 'system', 'content': self.persona}, *history,
                    {'role': 'user', 'content': text}]
        while len(messages) > 2 and not self._fits(messages, self.config['max_tokens']):
            messages = [messages[0], *messages[3:]]
        self._check_context(messages, self.config['max_tokens'])
        stream = self._load().create_chat_completion(
            messages=messages, stream=True,
            temperature=0.4, max_tokens=self.config['max_tokens'])
        buffer = ''
        try:
            for part in stream:
                if self.cancelled():
                    break
                buffer += part['choices'][0].get('delta', {}).get('content', '') or ''
                match = re.search(r'[.!?](?:\s|$)', buffer)
                if match or len(buffer) >= 300:
                    end = match.end() if match else len(buffer)
                    yield buffer[:end].strip()
                    buffer = buffer[end:]
            if buffer.strip() and not self.cancelled():
                yield buffer.strip()
        finally:
            stream.close()

    def _fits(self, messages, reserved):
        model = self._load()
        # Conservative per-message template allowance, plus generation budget.
        count = 64 + reserved + sum(24 + len(model.tokenize(m['content'].encode('utf-8'), add_bos=False)) for m in messages)
        return count <= self.config.get('context', 1024)

    def _check_context(self, messages, reserved):
        if not self._fits(messages, reserved):
            raise Refusal('That request exceeds my local context budget; please shorten it, sir.')

    def close(self):
        if self.model is not None:
            self.model.close()
            self.model = None

    psapi = ctypes.WinDLL('psapi', use_last_error=True)

"""The only dispatcher; fail closed, redacted auditing, bounded confirmations."""
from collections import deque
from copy import deepcopy
import time
from jsonschema import Draft202012Validator
from jsonschema.exceptions import ValidationError
from safety import Refusal
from tools.registry import REGISTRY


class Policy:
    def __init__(self, audit, *, dry_run=True, rate=20, failure_limit=5,
                 confirmation_seconds=8, clock=time.monotonic, registry=REGISTRY,
                 guard=None, mode=lambda: 'OFFLINE', deadline_seconds=5, on_timeout=lambda: None):
        self.audit, self.dry_run, self.registry = audit, dry_run, registry
        self.rate, self.failure_limit = rate, failure_limit
        self.confirmation_seconds, self.clock = confirmation_seconds, clock
        self.calls, self.failures, self.pending = deque(), 0, None
        self.guard, self.mode = guard, mode
        self.recent = deque(maxlen=6)
        self.deadline_seconds, self.on_timeout = deadline_seconds, on_timeout

    def refuse(self, message):
        self.failures += 1
        self.audit.record('refused', {}, 'refused')
        raise Refusal(message)

    def dispatch(self, call):
        return self._dispatch(call, confirmed=False)

    def _dispatch(self, call, *, confirmed):
        if self.failures >= self.failure_limit:
            raise Refusal('Safety lock engaged. Restart required, sir.')
        if not isinstance(call, dict) or set(call) != {'tool', 'args'} or not isinstance(call['tool'], str):
            return self.refuse('Invalid tool call.')
        name, args = call['tool'], call['args']
        tool = self.registry.get(name)
        if tool is None or tool.tier not in (0, 1):
            return self.refuse('That tool is unavailable, sir.')
        if getattr(tool, 'online', False) and self.mode() != 'ONLINE-RESTRICTED':
            return self.refuse('That tool requires confirmed Online mode, sir.')
        try:
            Draft202012Validator(tool.schema).validate(args)
        except ValidationError:
            return self.refuse('Invalid tool arguments.')
        now = self.clock()
        while self.calls and now - self.calls[0] >= 60:
            self.calls.popleft()
        if len(self.calls) >= self.rate:
            return self.refuse('Tool rate limit reached, sir.')
        if tool.tier == 1 and not confirmed:
            self.pending = (deepcopy(call), now + self.confirmation_seconds)
            self.audit.record(name, args, 'confirmation_requested', tool.tier)
            detail = ', '.join(f'{key}: {value}' for key, value in args.items())
            return f'Shall I perform {name}' + (f' ({detail})' if detail else '') + ', sir? Say yes or cancel within eight seconds.'
        self.calls.append(now)
        self.audit.record(name, args, 'dry_run' if self.dry_run else 'attempt', tool.tier)
        if self.dry_run:
            return f'I would perform {name}, sir. Dry run is enabled.'
        try:
            if self.guard:
                self.guard(name)
            from watchdog import Deadline
            with Deadline(self.deadline_seconds, self._timeout) as deadline:
                result = tool.handler(deepcopy(args))
            if deadline.expired:
                raise Refusal('Tool deadline exceeded; safety lock engaged, sir.')
        except Exception as exc:
            self.failures += 1
            self.audit.record(name, args, 'error', tool.tier)
            raise Refusal('The tool failed safely, sir.') from exc
        self.audit.record(name, args, 'ok', tool.tier)
        self.recent.append(name)
        return result

    def _timeout(self):
        self.failures = self.failure_limit
        self.on_timeout()

    def expire_confirmation(self):
        if self.pending and self.clock() >= self.pending[1]:
            self.pending = None
            self.audit.record('confirmation', {}, 'expired')
            return True
        return False

    def confirm(self, answer):
        pending, self.pending = self.pending, None
        if not pending or self.clock() >= pending[1] or answer.strip().lower().strip('.!?') != 'yes':
            return 'Cancelled, sir.'
        return self._dispatch(pending[0], confirmed=True)

class Runtime:
    def __init__(self, root, config, debug=False):
        self.root, self.config, self.debug = root, config, debug
        self.events, self.work = Queue(maxsize=64), Queue(maxsize=1)
        self.quit_event, self.cancelled = Event(), Event()
        self.epoch, self.busy, self.ptt = 0, False, False
        self.last_use, self.last_health = time.monotonic(), time.monotonic()
        self.lock = InstanceLock(Path(config['log_dir']) / 'jarvis.lock')
        self.lock.acquire()
        self.tray = self.speech = self.audio = self.browser = self.listener = self.memory_job = None
        try:
            self_check(config, root)
            self.memory_job = process_memory_limit(PROFILES[config['profile']][1])
            self.audit = Audit(Path(config['log_dir']) / 'audit.jsonl')
            self.health = Health(config)
            self.memory = Memory(config['memory_chars'])
            self.tray = Tray(self.kill)
            self.audio = Audio(self.post, self.state)
            self.speech = Speech(self.audio, config['tts'], self.post)
            self.session = SessionTools(self.memory, self.speech.say, self.speech.stop)
            self.browser = Chrome(config['browser'], Path(config['approved_folder']) / 'chrome-profiles',
                                  self.speech.say, self.audit)
            self.audit.mode = lambda: self.browser.mode.name
            registry = full_registry(FileTools(SecureFiles(config['approved_folder'])),
                                     SystemTools(config['apps']), self.session, self.browser)
            self.policy = Policy(self.audit, dry_run=config['dry_run'], registry=registry,
                rate=config['max_calls_per_minute'], failure_limit=config['failure_limit'],
                confirmation_seconds=config['confirmation_seconds'], guard=self._guard,
                mode=lambda: self.browser.mode.name, deadline_seconds=config['tool_deadline_seconds'],
                on_timeout=lambda: self.post('tool_timeout'))
            self.session.policy = self.policy
            self.whisper = Whisper(config['stt'], config['stt_confidence'])
            self.llm = LocalLLM(config['model'], (root / 'persona.txt').read_text(), registry,
                                cancelled=lambda: self.cancelled.is_set() or self.quit_event.is_set())
            self.assistant = Assistant(Intent(yaml.safe_load((root / 'commands.yaml').read_text())),
                                       self.policy, self.llm, self.memory, self.speech.say)
            self.conversation = Conversation(lambda: self.audio.open(True),
                lambda: self.audio.close(discard=True), timeout=config['conversation_seconds'])
            self.worker = Thread(target=self._worker, name='jarvis-inference', daemon=True)
            self.worker.start()
            self._hotkeys()
        except Exception:
            self.cleanup()
            raise

    def post(self, event, payload=None):
        try:
            self.events.put_nowait((event, payload))
        except Full:
            # Losing a control event is unsafe: immediately close the microphone.
            self.quit_event.set()
            if self.audio:
                self.audio.live = False

    def state(self, state):
        if self.tray:
            self.tray.state(state, self.browser.mode.name if self.browser else 'OFFLINE')

    def _guard(self, name):
        if self.cancelled.is_set() or self.quit_event.is_set():
            raise Refusal('Cancelled, sir.')
        self.health.tool(name)

    def _hotkeys(self):
        from pynput.keyboard import Listener, HotKey
        keys = self.config['hotkeys']
        ptt_keys = set(HotKey.parse(keys['push_to_talk']))
        pressed = set()
        combinations = [HotKey(HotKey.parse(keys['conversation']), lambda: self.post('conversation')),
                        HotKey(HotKey.parse(keys['stop']), self.interrupt),
                        HotKey(HotKey.parse(keys['quit']), self.kill)]
        def press(key):
            key = self.listener.canonical(key)
            was_active = ptt_keys <= pressed
            pressed.add(key)
            if not was_active and ptt_keys <= pressed:
                self.interrupt()
                self.ptt = True
                self.post('ptt_open')
            for combination in combinations:
                combination.press(key)
        def release(key):
            key = self.listener.canonical(key)
            was_active = ptt_keys <= pressed
            pressed.discard(key)
            if was_active and not ptt_keys <= pressed:
                self.ptt = False
                self.post('utterance')
            for combination in combinations:
                combination.release(key)
        self.listener = Listener(on_press=press, on_release=release)
        self.listener.start()

    def interrupt(self):
        self.epoch += 1
        self.cancelled.set()
        if self.speech:
            self.speech.stop()

    def kill(self, *_):
        self.quit_event.set()
        self.interrupt()
        if self.audio:
            self.audio.close(discard=True, notice=False)
        if self.browser:
            self.browser.mode.close()
        self.post('quit')

    def _worker(self):
        import pythoncom
        pythoncom.CoInitialize()
        try:
            while not self.quit_event.is_set():
                item = self.work.get()
                if item is None:
                    break
                epoch, pcm = item
                try:
                    self.health.heavy()
                    self.llm.close()
                    text = self.whisper.transcribe(pcm, lambda: epoch != self.epoch or self.quit_event.is_set())
                    self.whisper.close()
                    if epoch != self.epoch or self.quit_event.is_set():
                        continue
                    self.health.heavy()
                    answer = self.assistant.handle(text)
                    if self.policy.pending:
                        # Confirmation clock starts after the full spoken prompt.
                        self.policy.pending = (self.policy.pending[0], float('inf'))
                    if epoch == self.epoch and not self.quit_event.is_set() and not self.assistant.streamed:
                        self.speech.say(answer)
                except Refusal as exc:
                    if epoch == self.epoch:
                        self.speech.say(str(exc))
                except Exception:
                    if epoch == self.epoch:
                        self.speech.say('The request failed safely, sir.')
                finally:
                    self.whisper.close()
                    self.post('work_done')
        finally:
            pythoncom.CoUninitialize()

    def _capture_result(self):
        pcm = self.audio.close()
        if not pcm:
            return
        if self.busy:
            self.speech.say('The previous request is still stopping, sir.')
            return
        self.cancelled.clear()
        self.busy = True
        self.last_use = time.monotonic()
        self.conversation.touch()
        self.state('thinking')
        self.work.put_nowait((self.epoch, pcm))

    def _resume(self):
        if self.busy or not self.speech.idle() or self.quit_event.is_set():
            return
        if self.ptt:
            self.audio.open()
        elif self.policy.pending:
            if self.policy.pending[1] == float('inf'):
                self.policy.pending = (self.policy.pending[0], time.monotonic() + self.config['confirmation_seconds'])
            self.audio.open(True)
        elif self.conversation.enabled and not self.conversation.tick():
            self.audio.open(True)

    def loop(self):
        self.speech.say('Good evening, sir. JARVIS online in offline mode.')
        try:
            while not self.quit_event.is_set():
                now = time.monotonic()
                # No idle poll faster than five seconds; real deadlines wake exactly.
                deadlines = [self.last_health + 5, self.session.next_deadline()]
                if not self.busy and not self.audio.live:
                    deadlines.append(self.last_use + self.config['idle_exit_seconds'])
                if self.policy.pending:
                    deadlines.append(self.policy.pending[1])
                if self.conversation.enabled:
                    deadlines.append(self.conversation.deadline)
                if self.browser.mode.active:
                    deadlines.append(self.browser.mode.deadline)
                timeout = max(.01, min(deadlines) - now)
                try:
                    event, payload = self.events.get(timeout=timeout)
                except Empty:
                    event = 'tick'
                try:
                    if event == 'quit':
                        break
                    if event == 'conversation':
                        self.last_use = time.monotonic()
                        self.conversation.toggle(on_battery=self.health.on_battery(),
                            allow_battery=self.config['conversation_on_battery'])
                    elif event == 'ptt_open':
                        self.last_use = time.monotonic()
                        self._resume()
                    elif event == 'utterance':
                        self._capture_result()
                    elif event == 'audio_error':
                        self.audio.close(discard=True)
                        self.speech.say('Audio capture failed; please repeat, sir.')
                    elif event == 'speech_error':
                        self.audio.close(discard=True)
                        self.state('error')
                        raise Refusal('Speech output failed. Voice mode must stop safely.')
                    elif event == 'tool_timeout':
                        self.interrupt()
                        self.audio.close(discard=True)
                        self.speech.say('A tool exceeded its deadline. Safety lock engaged, sir.')
                    elif event == 'work_done':
                        self.busy = False
                        self._resume()
                    elif event == 'speech_done':
                        self._resume()
                    self._tick()
                except Refusal as exc:
                    if event == 'speech_error':
                        self.kill()
                    else:
                        self.speech.say(str(exc))
        finally:
            self.cleanup()

    def _tick(self):
        now = time.monotonic()
        self.conversation.tick()
        self.browser.mode.tick()
        self.session.tick()
        if self.policy.expire_confirmation():
            self.audio.close(discard=True)
            self.speech.say('Cancelled, sir.')
        if now - self.last_health >= 5:
            self.last_health = now
            if self.health.on_battery() and self.conversation.enabled and not self.config['conversation_on_battery']:
                self.conversation.close()
                self.speech.say('On battery; conversation mode closed, sir.')
        if not self.busy and now - self.last_use >= self.config['model_idle_seconds']:
            self.llm.close()
            self.whisper.close()
            self.speech.unload()
        if not self.busy and not self.audio.live and now - self.last_use >= self.config['idle_exit_seconds']:
            self.kill()

    def cleanup(self):
        self.quit_event.set()
        self.cancelled.set()
        if self.listener:
            self.listener.stop()
        if self.audio:
            self.audio.close(discard=True, notice=False)
        if self.browser:
            self.browser.mode.close()
        if self.speech:
            self.speech.stop()
            self.speech.say('Goodbye, sir.')
            end = time.monotonic() + .3
            while not self.speech.idle() and time.monotonic() < end:
                Event().wait(.01)
            self.speech.shutdown()
        if self.tray:
            self.tray.close()
        if hasattr(self, 'memory'):
            self.memory.clear()
        self.lock.close()
        if self.memory_job:
            self.memory_job.Close()

"""Conservative guards. No file mutation or OS action is implemented here."""
import hashlib
import os
from pathlib import Path
import re
import stat


class Refusal(ValueError):
    pass


def user_text(text):
    if not isinstance(text, str) or not 0 < len(text) <= 2000:
        raise Refusal('Input must contain 1 to 2000 characters.')
    if any(ord(c) < 32 for c in text):
        raise Refusal('Control characters are not permitted.')
    if re.search(r'\b(delete|remove|rmdir|trash|format|install|uninstall|download|'
                 r'powershell|cmd|terminal|registry|password|cvv|otp|overwrite|'
                 r'rename|move|purchase|ignore|bypass)\b', text, re.I):
        raise Refusal('That request is outside my permissions, sir.')
    if any(c in text for c in (';', '|', '`', '$(', '\x00')):
        raise Refusal('Shell-like input is not permitted.')
    return text


def no_links(path):
    path = Path(os.path.abspath(path))
    for part in (*reversed(path.parents), path):
        if part.is_symlink():
            raise Refusal('Symbolic links are not permitted.')
        if part.exists():
            info = part.lstat()
            if getattr(info, 'st_file_attributes', 0) & stat.FILE_ATTRIBUTE_REPARSE_POINT:
                raise Refusal('Reparse points are not permitted.')
    return path


def safe_path(root, name, *, new=False, max_bytes=65536):
    root = no_links(Path(root).expanduser())
    # Deliberately restrict Phase 1 to a single flat folder and plain text names.
    if not isinstance(name, str) or not re.fullmatch(r'[A-Za-z0-9][A-Za-z0-9 _-]{0,60}\.txt', name):
        raise Refusal('Only simple .txt names are permitted.')
    if re.search(r'(secret|password|credential|token|private|key)', name, re.I):
        raise Refusal('Sensitive paths are not permitted.')
    if name.split('.')[0].upper() in {'CON', 'PRN', 'AUX', 'NUL', *('COM'+str(i) for i in range(10)), *('LPT'+str(i) for i in range(10))}:
        raise Refusal('Reserved filename.')
    target = no_links(root / name)
    if target.resolve().parent != root.resolve():
        raise Refusal('Path escapes the approved folder.')
    if target.exists():
        if new or not target.is_file() or target.stat().st_nlink != 1:
            raise Refusal('Existing files, hard links or non-files are not allowed here.')
        if target.stat().st_size > max_bytes:
            raise Refusal('File exceeds the size limit.')
    return target


def verify_model(path, expected, max_bytes):
    path = no_links(path)
    if not isinstance(expected, str) or not re.fullmatch('[0-9a-f]{64}', expected):
        raise Refusal('A trusted model SHA256 must be configured.')
    if not path.is_file() or path.stat().st_size > max_bytes:
        raise Refusal('Model missing or exceeds the disk allowance.')
    with path.open('rb') as stream:
        actual = hashlib.file_digest(stream, 'sha256').hexdigest()
    if actual != expected:
        raise Refusal('Model checksum mismatch.')
    return path


def guard_app(name, apps):

    raise Refusal('App launching is unavailable in Phase 1.')


def guard_typing(text, *, own_chrome=False):
    raise Refusal('Input control is disabled in Phase 1.')

def obj(**properties):

def checked(name):
    def decorate(handler):
        @wraps(handler)
        def wrapped(self, args):
            Draft202012Validator(SCHEMAS[name]).validate(args)
            return handler(self, args)
        return wrapped
    return decorate

"""Windows handle-based, no-delete-sharing file access; flat approved folder only."""
from contextlib import contextmanager
import os
from pathlib import Path
import shutil
from safety import Refusal, no_links, safe_path


class SecureFiles:
    def __init__(self, root, max_bytes=65536, minimum_free=100_000_000):
        self.root = no_links(Path(root).expanduser())
        self.max_bytes, self.minimum_free = max_bytes, minimum_free
        if not self.root.is_dir():
            raise Refusal('Create the approved AssistantFiles folder during setup.')

    @contextmanager
    def handle(self, name, *, write=False, new=False, image=False):
        if os.name != 'nt':
            raise Refusal('Secure file handles are implemented for Windows only.')
        import win32con as c
        import win32file as f
        from pywintypes import error
        if image:
            import re
            if not re.fullmatch(r'screenshot-[0-9a-f]{32}\.png', name):
                raise Refusal('Invalid screenshot name.')
            target = no_links(self.root / name)
        else:
            target = safe_path(self.root, name, new=new, max_bytes=self.max_bytes)
        if write and shutil.disk_usage(self.root).free < self.minimum_free:
            raise Refusal('Disk space is low, sir.')
        # Holding the root without FILE_SHARE_DELETE prevents rename/replacement.
        directory = f.CreateFile(str(self.root), c.GENERIC_READ, c.FILE_SHARE_READ | c.FILE_SHARE_WRITE,
                                 None, c.OPEN_EXISTING, c.FILE_FLAG_BACKUP_SEMANTICS | c.FILE_FLAG_OPEN_REPARSE_POINT, None)
        handle = None
        try:
            if f.GetFileInformationByHandle(directory)[0] & c.FILE_ATTRIBUTE_REPARSE_POINT:
                raise Refusal('Approved folder is a reparse point.')
            expected_root = f.GetFinalPathNameByHandle(directory, 0).casefold()
            rights = (c.GENERIC_READ | c.GENERIC_WRITE) if write else c.GENERIC_READ
            handle = f.CreateFile(str(target), rights, 0, None, c.CREATE_NEW if new else c.OPEN_EXISTING,
                                  c.FILE_FLAG_OPEN_REPARSE_POINT, None)
            info = f.GetFileInformationByHandle(handle)
            actual = f.GetFinalPathNameByHandle(handle, 0).casefold()
            if actual != expected_root.rstrip('\\') + '\\' + name.casefold():
                raise Refusal('Handle escaped the approved folder.')
            if info[0] & (c.FILE_ATTRIBUTE_REPARSE_POINT | c.FILE_ATTRIBUTE_DIRECTORY) or info[7] != 1:
                raise Refusal('Links and directories are not allowed.')
            if f.GetFileSize(handle) > (8_000_000 if image else self.max_bytes):
                raise Refusal('File is too large.')
            yield handle
        except error as exc:
            raise Refusal('File access was refused, sir.') from exc
        finally:
            if handle is not None:
                handle.Close()
            directory.Close()

    def read(self, name):
        import win32file as f
        with self.handle(name) as handle:
            size = f.GetFileSize(handle)
            data = f.ReadFile(handle, size)[1] if size else b''
        return data.decode('utf-8', errors='strict')

    def write(self, name, text, *, new=False):
        import win32file as f
        data = text.encode('utf-8')
        if not 0 < len(data) <= self.max_bytes:
            raise Refusal('Note exceeds the size limit.')
        with self.handle(name, write=True, new=new) as handle:
            if f.GetFileSize(handle) + len(data) > self.max_bytes:
                raise Refusal('Note exceeds the size limit.')
            f.SetFilePointer(handle, 0, 2)
            f.WriteFile(handle, data)
            f.FlushFileBuffers(handle)

    def image(self, name, data):
        import win32file as f
        if len(data) > 8_000_000:
            raise Refusal('Screenshot exceeds its size limit.')
        with self.handle(name, write=True, new=True, image=True) as handle:
            f.WriteFile(handle, data)
            f.FlushFileBuffers(handle)

    def names(self):
        result = []
        for path in self.root.iterdir():
            try:
                safe_path(self.root, path.name)
                if path.is_file():
                    result.append(path.name)
            except Refusal:
                continue
            if len(result) >= 100:
                break
        return sorted(result)

from collections import deque
from schemas import checked
from safety import Refusal
import time


class Memory:
    def __init__(self, max_chars=2400):
        self.turns, self.max_chars, self.last_volume = deque(maxlen=6), max_chars, None

    def remember(self, user, answer):
        self.turns.append((user[:600], answer[:600]))
        while sum(len(a)+len(b) for a, b in self.turns) > self.max_chars:
            self.turns.popleft()

    def clear(self):
        self.turns.clear()
        self.last_volume = None

    def messages(self):
        return [dict(role=role, content=text) for turn in self.turns
                for role, text in zip(('user', 'assistant'), turn)]


class SessionTools:
    def __init__(self, memory, speak, stop, clock=time.monotonic):
        self.memory, self.speak, self.stop, self.clock = memory, speak, stop, clock
        self.timers, self.policy = [], None

    @checked('set_timer')
    def set_timer(self, args):
        if len(self.timers) >= 10:
            raise Refusal('Ten reminders are already active, sir.')
        self.timers.append((self.clock() + args['seconds'], args['label']))
        return 'Reminder set for this session, sir.'

    def tick(self):
        due = [item for item in self.timers if item[0] <= self.clock()]
        self.timers = [item for item in self.timers if item[0] > self.clock()]
        for _, label in due:
            self.speak('Reminder, sir. ' + label)

    def next_deadline(self):
        return min((item[0] for item in self.timers), default=float('inf'))

    @checked('stop_speaking')
    def stop_speaking(self, args):
        self.stop()
        return 'Speech stopped, sir.'

    @checked('forget_conversation')
    def forget_conversation(self, args):
        self.memory.clear()
        return 'Conversation forgotten, sir.'

    @checked('what_did_you_do')
    def what_did_you_do(self, args):
        return 'Recent actions: ' + ', '.join(self.policy.recent) if self.policy and self.policy.recent else 'No recent actions, sir.'

"""Read-only OS backstop checks; never silently repair permissions or firewall."""
import os
from pathlib import Path
import sys
from safety import Refusal, no_links, verify_model
from health import PROFILES


class InstanceLock:
    def __init__(self, path):
        self.path, self.file = no_links(path), None

    def acquire(self):
        import msvcrt
        if not self.path.parent.is_dir():
            raise Refusal('The private log directory is missing; complete the limited-account setup first.')
        self.file = self.path.open('a+b')
        if self.path.stat().st_size == 0:
            self.file.write(b'0')
            self.file.flush()
        self.file.seek(0)
        try:
            msvcrt.locking(self.file.fileno(), msvcrt.LK_NBLCK, 1)
        except OSError as exc:
            self.file.close()
            self.file = None
            raise Refusal('Another JARVIS instance is running, sir.') from exc

    def close(self):
        if self.file:
            self.file.close()
            self.file = None


def check_firewall(executable):
    import win32com.client
    firewall = win32com.client.Dispatch('HNetCfg.FwPolicy2')
    for profile in (1, 2, 4):
        if not firewall.FirewallEnabled(profile):
            raise Refusal('Windows Firewall must be enabled on every profile.')
    expected = os.path.normcase(os.path.abspath(executable))
    covered = 0
    for rule in firewall.Rules:
        if (rule.Enabled and rule.Direction == 2 and rule.Action == 0
            and os.path.normcase(os.path.abspath(rule.ApplicationName or '.')) == expected
            and rule.Protocol == 256 and rule.LocalAddresses == '*' and rule.RemoteAddresses == '*'
            and rule.InterfaceTypes == 'All' and not rule.ServiceName):
            covered |= rule.Profiles
    if covered & 7 != 7:
        raise Refusal('An unrestricted outbound block for this exact Python executable is required.')


def check_account(config):
    import getpass
    import ctypes
    if getpass.getuser().casefold() != config['security']['account'].casefold():
        raise Refusal('Run JARVIS under its dedicated limited Windows account.')
    if ctypes.windll.shell32.IsUserAnAdmin():
        raise Refusal('JARVIS must never run as administrator.')
    home = Path(config['security']['protected_home'])
    if not home.is_absolute() or home == Path.home():
        raise Refusal('A separate protected personal home must be configured.')
    try:
        with os.scandir(home) as entries:
            next(entries, None)
    except PermissionError:
        return
    except OSError as exc:
        raise Refusal('Cannot establish the personal-home access boundary.') from exc
    raise Refusal('The assistant account still has access to the personal home.')


def check_private_folder(path):
    import win32security as s
    import win32api
    import win32con
    path = no_links(path)
    if not path.is_dir():
        raise Refusal('Required private directory is missing.')
    token = s.OpenProcessToken(win32api.GetCurrentProcess(), win32con.TOKEN_QUERY)
    try:

def check_deployment_root(root):
    """The assistant must not execute source from a cloud-synced or user-writable tree."""
    root = no_links(root)
    if not root.is_dir() or any(part.casefold() == 'onedrive' for part in root.parts):
        raise Refusal('Deploy JARVIS in a local, unsynced directory outside OneDrive.')
    import win32security as s
    import win32api
    import win32con
    token = s.OpenProcessToken(win32api.GetCurrentProcess(), win32con.TOKEN_QUERY)
    try:
        user = s.GetTokenInformation(token, s.TokenUser)[0]
    finally:
        token.Close()
    user_sid = s.ConvertSidToStringSid(user)
    descriptor = s.GetFileSecurity(str(root), s.DACL_SECURITY_INFORMATION)
    acl = descriptor.GetSecurityDescriptorDacl()
    if acl is None:
        raise Refusal('Application directory has an unrestricted ACL.')
    write_mask = (win32con.FILE_GENERIC_WRITE | win32con.FILE_WRITE_DATA | win32con.FILE_APPEND_DATA |
                  win32con.FILE_WRITE_ATTRIBUTES | win32con.FILE_WRITE_EA | win32con.WRITE_DAC |
                  win32con.WRITE_OWNER | win32con.DELETE)
    for index in range(acl.GetAceCount()):
        ace = acl.GetAce(index)
        if ace[0][0] == s.ACCESS_ALLOWED_ACE_TYPE and s.ConvertSidToStringSid(ace[2]) == user_sid and ace[1] & write_mask:
            raise Refusal('Assistant account must have read-and-execute only on the application directory.')

    import sounddevice as sd

    sd.check_output_settings(channels=1, dtype='int16', samplerate=16000)
    # Outbound probe exception has not been approved. Never silently bypass it.
    if not config['security'].get('network_probe_exception', False):
        raise Refusal('The required outbound self-test awaits resolution of the network exception.')
    raise Refusal('Outbound self-test implementation awaits explicit setup-exception approval.')

import gc
import math
from safety import Refusal, verify_model


class Whisper:
    def __init__(self, config, threshold=.80):
        self.config, self.threshold, self.model = config, threshold, None

    def transcribe(self, pcm, cancelled=lambda: False):
        if not pcm or len(pcm) > 16000 * 2 * 15:
            raise Refusal("I didn't catch that, sir.")
        import numpy as np
        from pywhispercpp.model import Model
        if self.model is None:
            path = verify_model(self.config['path'], self.config['sha256'], self.config['max_bytes'])
            self.model = Model(str(path), n_threads=self.config.get('threads', 1),
                               context_params={'use_gpu': False}, language='en', no_context=True,
                               print_realtime=False, print_progress=False, print_timestamps=False)
        samples = np.frombuffer(pcm, dtype=np.int16).astype(np.float32) / 32768
        segments = self.model.transcribe(samples, extract_probability=True, abort_callback=cancelled)
        if cancelled() or not segments:
            raise Refusal("I didn't catch that, sir.")
        confidence = min(float(getattr(item, 'probability', float('nan'))) for item in segments)
        if not math.isfinite(confidence) or confidence < self.threshold:
            raise Refusal("I didn't catch that confidently, sir. Please repeat.")
        text = ' '.join(item.text for item in segments).strip()
        if not text:
            raise Refusal("I didn't catch that, sir.")
        return text

    def close(self):
        if self.model is not None:
            self.model = None
            gc.collect()

@dataclass(frozen=True)
class Tool:
    handler: object
    schema: dict
    tier: int = 0
    online: bool = False


REGISTRY = MappingProxyType({
    'get_time_date': Tool(get_time_date, EMPTY),
    'get_battery_status': Tool(get_battery_status, EMPTY),
    'system_status': Tool(system_status, EMPTY),
})


def call_schema(registry=REGISTRY):
    variants = [{'type': 'object', 'properties': {'tool': {'const': name}, 'args': tool.schema},
                 'required': ['tool', 'args'], 'additionalProperties': False}
                for name, tool in registry.items()]
    variants.append({'type': 'object', 'properties': {'tool': {'const': 'none'}, 'route': {'const': 'chat'}},
                     'required': ['tool', 'route'], 'additionalProperties': False})
    return {'oneOf': variants}


def full_registry(files, system, session, browser=None):
    from schemas import SCHEMAS
    registry = dict(REGISTRY)
    for owner, names, tier in [
        (files, ('list_folder', 'read_file'), 0),
        (files, ('create_note', 'append_note', 'undo_last_note_action', 'take_screenshot'), 1),
        (system, ('open_app', 'close_app', 'set_volume', 'mute', 'set_brightness', 'media_play_pause',
                  'media_next', 'media_prev', 'lock_screen'), 0),
        (session, ('set_timer', 'stop_speaking', 'forget_conversation', 'what_did_you_do'), 0),
    ]:
        for name in names:
            registry[name] = Tool(getattr(owner, name), SCHEMAS[name], tier)
    if browser:
        registry['enable_online'] = Tool(browser.enable_online, SCHEMAS['enable_online'], 1)
        for name in ('open_chrome', 'web_search', 'close_chrome', 'type_text'):
            registry[name] = Tool(getattr(browser, name), SCHEMAS[name], 1 if name == 'type_text' else 0, True)
    return MappingProxyType(registry)

DENIED_APPS = {'cmd', 'powershell', 'pwsh', 'wt', 'wscript', 'cscript', 'mshta', 'rundll32',
               'reg', 'regedit', 'python', 'pythonw', 'node', 'bash', 'sh', 'chrome', 'msedge',

def approved_executable(name, apps):
    item = apps.get(name)
    if not item or set(item) != {'path', 'sha256'}:
        raise Refusal('That app is not on the approved list, sir.')
    path = Path(item['path'])
    if not path.is_absolute() or path.suffix.lower() != '.exe' or path.stem.lower() in DENIED_APPS:
        raise Refusal('Browsers, shells and interpreters are not app tools.')
    return verify_model(path, item['sha256'], 500_000_000)


class SystemTools:
    def __init__(self, apps):
        self.apps, self.launched = apps, {}

    @checked('open_app')
    def open_app(self, args):
        name = args['app']
        exe = approved_executable(name, self.apps)
        old = self.launched.get(name)
        if old and old.poll() is None:
            return 'That app is already open, sir.'
        # argv is fixed; user/model text cannot become process flags.
        process = subprocess.Popen([str(exe)], shell=False, close_fds=True, cwd=str(exe.parent))
        self.launched[name] = process
        return 'Opening ' + name + ', sir.'

    @checked('close_app')
    def close_app(self, args):
        process = self.launched.get(args['app'])
        if process is None or process.poll() is not None:
            raise Refusal('I did not launch a running instance of that app, sir.')
        graceful_close(process.pid)
        return 'I requested that my app instance close, sir.'

"""Bounded sentence queue and cancellable RAM playback, with a local OS fallback."""
from queue import Queue, Empty, Full
from threading import Event, RLock, Thread
import time
from safety import Refusal, verify_model


class Speech:
    def __init__(self, audio, config, post, *, backend=None, player=None):
        self.audio, self.config, self.post = audio, config, post
        self.queue, self.closed = Queue(maxsize=12), Event()
        self.lock, self.generation, self.active = RLock(), 0, False
        self.backend, self.player, self.voice, self.engine = backend, player, None, None
        self.output_stream = None
        self.cache, self.cache_bytes = {}, 0
        self.worker = Thread(target=self._worker, name='jarvis-speech', daemon=True)
        self.worker.start()

    def say(self, text):
        if not text or self.closed.is_set():
            return
        item = (self.generation, str(text)[:65536])
        while not self.closed.is_set() and item[0] == self.generation:
            try:
                self.queue.put(item, timeout=.1)
                return
            except Full:
                continue

    def stop(self):
        with self.lock:
            self.generation += 1
            stream = self.output_stream
        if stream:
            try:
                stream.abort()
            except Exception:
                pass
        if self.engine:
            # Engine loop observes generation on its next 10 ms pump.
            pass
        while True:
            try:
                self.queue.get_nowait()
                self.queue.task_done()
            except Empty:
                break

    def _chunks(self, text):
        if text in self.cache:
            yield from self.cache[text]
            return
        canned = text in ('Goodbye, sir.', 'Cancelled, sir.', "I didn't catch that, sir.",
                          'Good evening, sir. JARVIS online in offline mode.')
        cached, size = [], 0
        for chunk in self._uncached_chunks(text):
            if canned:
                size += len(chunk[0])
                if size <= 1_000_000 - self.cache_bytes:
                    cached.append(chunk)
                else:
                    canned = False
                    cached.clear()
            yield chunk
        if canned:
            self.cache[text] = cached
            self.cache_bytes += size

    def _uncached_chunks(self, text):
        if self.backend:
            yield from self.backend(text)
            return
        if self.voice is None:
            import json
            import onnxruntime as ort
            from piper import PiperVoice
            from piper.config import PiperConfig
            path = verify_model(self.config['path'], self.config['sha256'], self.config['max_bytes'])
            verify_model(self.config['config_path'], self.config['config_sha256'], 100000)
            with open(self.config['config_path'], encoding='utf-8') as stream:
                voice_config = json.load(stream)
            if voice_config.get('phoneme_type', 'espeak') != 'espeak' or not voice_config.get('espeak', {}).get('voice', '').startswith('en'):
                raise Refusal('Only a verified local English espeak voice is supported.')
            options = ort.SessionOptions()
            options.intra_op_num_threads = 1
            options.inter_op_num_threads = 1
            options.enable_mem_pattern = False
            ort.disable_telemetry_events()
            self.voice = PiperVoice(config=PiperConfig.from_dict(voice_config),
                session=ort.InferenceSession(str(path), sess_options=options, providers=['CPUExecutionProvider']))
            self.voice.use_tashkeel = False
        for chunk in self.voice.synthesize(text):
            yield chunk.audio_int16_bytes, chunk.sample_rate, chunk.sample_channels

    def _play(self, data, rate, channels, generation):
        if self.player:
            self.player(data, rate, channels, lambda: generation != self.generation)
            return
        import sounddevice as sd
        with sd.RawOutputStream(samplerate=rate, channels=channels, dtype='int16', blocksize=max(1, rate // 50)) as stream:
            with self.lock:
                self.output_stream = stream
            step = max(2 * channels, int(rate * .02) * 2 * channels)
            try:
                for offset in range(0, len(data), step):
                    if generation != self.generation or self.closed.is_set():
                        break
                    stream.write(data[offset:offset+step])
            finally:
                with self.lock:
                    self.output_stream = None

    def _fallback(self, text, generation):
        import pyttsx3
        if self.engine is None:
            self.engine = pyttsx3.init()
            self.engine.setProperty('rate', 170)
        self.engine.say(text)
        self.engine.startLoop(False)
        try:
            while self.engine.isBusy() and generation == self.generation and not self.closed.is_set():
                self.engine.iterate()
                self.closed.wait(.01)
            self.engine.stop()
        finally:
            self.engine.endLoop()

    def _worker(self):
        while not self.closed.is_set():
            item = self.queue.get()
            if item is None:
                self.queue.task_done()
                break
            generation, text = item
            try:
                if generation != self.generation:
                    continue
                self.active = True
                self.audio.begin_speech()
                if self.config.get('backend') == 'system':
                    self._fallback(text, generation)
                else:
                    for data, rate, channels in self._chunks(text):
                        if generation != self.generation:
                            break
                        self._play(data, rate, channels, generation)
            except Exception:
                self.post('speech_error')
            finally:
                self.active = False
                self.audio.end_speech()
                self.queue.task_done()
                self.post('speech_done')

    def idle(self):
        return not self.active and self.queue.unfinished_tasks == 0

    def unload(self):
        if self.idle():
            self.voice = None
            self.cache.clear()
            self.cache_bytes = 0

    def shutdown(self):
        self.stop()
        self.closed.set()
        self.queue.put(None)
        self.worker.join(timeout=.3)

"""Deadline notification never pretends to forcibly cancel a native call."""
from threading import Timer
import time


class Deadline:
    def __init__(self, seconds, on_timeout):
        self.seconds, self.on_timeout = seconds, on_timeout
        self.timer, self.expired = None, False

    def __enter__(self):
        def expire():
            self.expired = True
            self.on_timeout()
        self.timer = Timer(self.seconds, expire)
        self.timer.daemon = True
        self.timer.start()
        return self

    def __exit__(self, *_):
        self.timer.cancel()


def process_memory_limit(max_bytes):
    import win32api
    import win32job
    job = win32job.CreateJobObject(None, None)
    info = win32job.QueryInformationJobObject(job, win32job.JobObjectExtendedLimitInformation)
    info['BasicLimitInformation']['LimitFlags'] = win32job.JOB_OBJECT_LIMIT_PROCESS_MEMORY
    info['ProcessMemoryLimit'] = max_bytes
    win32job.SetInformationJobObject(job, win32job.JobObjectExtendedLimitInformation, info)
    win32job.AssignProcessToJobObject(job, win32api.GetCurrentProcess())
    return job
